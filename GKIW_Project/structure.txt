├── game/                   # Logika samej gry w Go (niezależna od grafiki 3D)
│   ├── __init__.py

├── render/                 # Klasy pomagające w rysowaniu obiektów

│   └── board_renderer.py   # Klasa dbająca o rysowanie planszy i kamieni na odpowiednich pozycjach
│
├── utils/                  # Funkcje pomocnicze

│
├── main.py                 # Punkt wejścia (tylko inicjalizuje okno i uruchamia grę)
